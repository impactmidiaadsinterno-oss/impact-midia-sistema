CREATE TABLE clients (id uuid DEFAULT gen_random_uuid() PRIMARY KEY, user_id uuid, company_name text NOT NULL, contact_name text, monthly_value numeric DEFAULT 0, status text, relationship text, results text, churn_risk text, health_score int, niche text, city text, start_date date, due_day int, payment_method text, email text, whatsapp text, notes text, created_at timestamptz DEFAULT now()); CREATE TABLE revenue_entries (id uuid DEFAULT gen_random_uuid() PRIMARY KEY, user_id uuid, client_id uuid, client_name text, amount numeric, paid_at date, reference_month text, status text, payment_method text, notes text, created_at timestamptz DEFAULT now()); CREATE TABLE expense_entries (id uuid DEFAULT gen_random_uuid() PRIMARY KEY, user_id uuid, description text, amount numeric, date date, category text, type text, notes text, created_at timestamptz DEFAULT now()); CREATE TABLE goals (id uuid DEFAULT gen_random_uuid() PRIMARY KEY, user_id uuid, title text, type text, target_value numeric, current_value numeric DEFAULT 0, month int, year int, deadline date, status text, created_at timestamptz DEFAULT now()); CREATE TABLE meetings (id uuid DEFAULT gen_random_uuid() PRIMARY KEY, user_id uuid, company text, date date, responsible text, closed boolean DEFAULT false, reason text, notes text, created_at timestamptz DEFAULT now()); CREATE TABLE monthly_history (id uuid DEFAULT gen_random_uuid() PRIMARY KEY, user_id uuid, month text UNIQUE, revenue numeric DEFAULT 0, expenses numeric DEFAULT 0, clients int DEFAULT 0, new_clients int DEFAULT 0, churns int DEFAULT 0); -- Enable RLS on all tables ALTER TABLE clients ENABLE ROW LEVEL SECURITY; ALTER TABLE revenue_entries ENABLE ROW LEVEL SECURITY;
